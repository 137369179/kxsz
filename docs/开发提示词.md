> ## T4. HanziEngine 笔顺方向角验证
>
> ### 现状
> `src/utils/hanziEngine.js` 第 310-360 行 `checkTraceAccuracy()` 仅端点距离判定。
>
> ### 改造目标
> 引入方向角验证，5-6 岁容差 60°，7-8 岁容差 45°。
>
> ### 实施步骤
>
> ```javascript
> // src/utils/hanziEngine.js 新增方法：
> strokeDirectionValidator(userPath, expected, toleranceDeg = 60) {
>   if (!userPath || userPath.length < 3) return false;
>
>   // 计算用户轨迹平均方向
>   const userAngle = this._avgPathAngle(userPath);
>   // 计算期望笔画方向
>   const expectedAngle = Math.atan2(
>     expected.end.y - expected.start.y,
>     expected.end.x - expected.start.x
>   ) * 180 / Math.PI;
>
>   // 容差对比
>   let diff = Math.abs(userAngle - expectedAngle);
>   if (diff > 180) diff = 360 - diff;
>   return diff < toleranceDeg;
> }
>
> _avgPathAngle(path) {
>   let sum = 0;
>   let count = 0;
>   for (let i = 1; i < path.length - 1; i++) {
>     const dx = path[i].x - path[i-1].x;
>     const dy = path[i].y - path[i-1].y;
>     if (Math.abs(dx) + Math.abs(dy) > 0.01) {
>       sum += Math.atan2(dy, dx);
>       count++;
>     }
>   }
>   return count > 0 ? (sum / count) * 180 / Math.PI : 0;
> }
>
> // checkTraceAccuracy() 末尾追加：
> checkTraceAccuracy(userPath, expectedStroke) {
>   const startOk = this._dist(userPath[0], expectedStroke.start) < this.tolerance;
>   const endOk = this._dist(userPath[userPath.length-1], expectedStroke.end) < this.tolerance;
>   const age = ebbinghausManager.getAge();
>   const tolerance = age <= 6 ? 60 : 45;
>   const dirOk = this.strokeDirectionValidator(userPath, expectedStroke, tolerance);
>   return startOk && endOk && dirOk;
> }
> ```
>
> ### 验收标准
> - [ ] 单元测试 `tests/unit/hanziEngine.test.js` 验证绕路轨迹不通过
> - [ ] 手动测试：从相反方向画笔画 → 显示"笔顺好像不太对"
